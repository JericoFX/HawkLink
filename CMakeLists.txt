if(MSVC AND CMAKE_GENERATOR MATCHES "Ninja"
        AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX STREQUAL "Nota: inclusi├│n del archivo: ")
    # This installed compiler emits the Spanish prefix as UTF-8. CMake's
    # cached OEM-decoded value lost header dependencies; the byte probe in
    # implementation/evidence/msvc-include-prefix confirms the actual prefix.
    set(CMAKE_CL_SHOWINCLUDES_PREFIX "Nota: inclusión del archivo: ")
endif()

    target_compile_definitions(hawken_foundation PUBLIC NOMINMAX WIN32_LEAN_AND_MEAN)

    target_compile_options(hawken_foundation PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(WIN32)
    target_link_libraries(hawken_foundation PRIVATE bcrypt ws2_32)
else()
    find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
    target_link_libraries(hawken_foundation PRIVATE OpenSSL::Crypto)
